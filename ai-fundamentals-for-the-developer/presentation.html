write(f, d)  -> os.write(f, d)
flush(f)     -> os.flush(f)
unlock(f)    -> os.unlock(f)
close(f)     -> os.close(f)

Deep code

save(path, data):
  f = os.open(path)
  for attempt in 1..3:
    for attempt in 1..3:
      if os.lock(f):
        os.write(f, data)
        os.flush(f)
        os.unlock(f)
        break
  os.close(f)
